cmake_minimum_required(VERSION 3.31.6)
project(fairllm LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS fairllm_compile=2 fairllm_link=1)
set(CMAKE_JOB_POOL_COMPILE fairllm_compile)
set(CMAKE_JOB_POOL_LINK fairllm_link)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(Vulkan_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/vulkan-headers/include" CACHE PATH "" FORCE)
set(SPIRV-Headers_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake" CACHE PATH "" FORCE)
if(CMAKE_HOST_WIN32)
    set(FAIRLLM_HOST "windows-x86_64")
    set(FAIRLLM_EXE ".exe")
else()
    set(FAIRLLM_HOST "linux-x86_64")
    set(FAIRLLM_EXE "")
endif()
set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/${FAIRLLM_HOST}/glslc${FAIRLLM_EXE}" CACHE FILEPATH "" FORCE)
if(DEFINED ENV{FAIRLLM_HOST_TOOLCHAIN})
    set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "$ENV{FAIRLLM_HOST_TOOLCHAIN}" CACHE FILEPATH "" FORCE)
endif()
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_TOOLS_INSTALL OFF CACHE BOOL "" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run python scripts/fetch-llama.py before building FairLLM")
endif()
add_subdirectory("${LLAMA_SOURCE}" "llama")
target_include_directories(ggml-vulkan PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/spirv-headers/include")
set(PROJECT_SOURCE_DIR "${LLAMA_SOURCE}")
add_subdirectory("${LLAMA_SOURCE}/tools/ui" "llama-ui")
add_subdirectory("${LLAMA_SOURCE}/tools/server" "llama-server")
add_library(fairllm SHARED native-engine.cpp)
target_link_libraries(fairllm PRIVATE llama-server-impl log)
target_link_options(fairllm PRIVATE "-Wl,-z,max-page-size=16384")
